    //io 객체: 클라이언트와 실시간 양방향 통신을 가능하게 하는 객체, 서버에서 클라이언트로 이벤트를 보내거나 클라이언트에서 서버로 이벤트를 받을 때 사용

    const initSocket = (io) => {
        // io 객체를 초기화하는 함수, app.js에서 서버 생성 후 이 함수를 호출해서 io 객체를 전달
        _io = io;
        io.on('connection', (socket) => {
            // 클라이언트가 연결되었을 때 실행되는 이벤트 리스너, socket 객체는 연결된 클라이언트와의 통신에 사용
            const user = socket.handshake.auth.user;
            // 클라이언트가 연결할 때 보낸 인증 정보(로그인한 사용자 정보)에서 user 객체를 가져옴
            if (!user) {
                // user 정보가 없으면 연결 종료 (로그인 안 한 사용자는 소켓 연결 못 하도록)
                return socket.disconnect();
            }
            socket.join(`유저: ${user._id}`);
            socket.data.user = user;
            console.log(`[소켓 연결] 유저: ${user.name} / 룸: 유저: ${user._id}`);
        });
    };

    const emitToUser = (userId, event, data) => {
        const room = _io.sockets.adapter.rooms.get(`유저: ${userId}`);
        // room이 없으면 해당 유저가 현재 접속 중이 아닌 것

        _io.to(`유저: ${userId}`).emit(event, data);
        //      ↑ 룸 지정          ↑ 이벤트명  ↑ 데이터
    };

    module.exports = {
        initSocket,
        emitToUser,
    };

#3 notificationRouter.js
const express = require('express');
const router = express.Router();
const notificationService = require('../services/notificationService');

// 로그인 확인 미들웨어
function isLoggedIn(req, res, next) {
    if (req.isAuthenticated()) {
        return next();
    }
    return res.status(401).json({ error: '로그인이 필요합니다.' });
}

// 알림 목록 조회
router.get('/', isLoggedIn, async (req, res) => {
    try {
        const notifications = await notificationService.getNotifications(req.user._id);
        res.json({ notifications });
    } catch (err) {
        res.status(500).json({ error: '알림 조회 실패' });
    }
});

// 읽지 않은 알림 수 조회
router.get('/unread-count', isLoggedIn, async (req, res) => {
    try {
        const count = await notificationService.getUnreadCount(req.user._id);
        res.json({ count });
    } catch (err) {
        res.status(500).json({ error: '알림 수 조회 실패' });
    }
});

// 알림 전체 읽음 처리
router.post('/read-all', isLoggedIn, async (req, res) => {
    try {
        await notificationService.markAllAsRead(req.user._id);
        res.json({ success: true });
    } catch (err) {
        res.status(500).json({ error: '읽음 처리 실패' });
    }
});

// 알림 단건 읽음 처리
router.post('/read/:id', isLoggedIn, async (req, res) => {
    try {
        await notificationService.markAsRead(req.params.id, req.user._id);
        res.json({ success: true });
    } catch (err) {
        res.status(500).json({ error: '읽음 처리 실패' });
    }
});

module.exports = router;


#4 notificationService
const Notification = require('../models/Notification');
const { emitToUser } = require('../config/socket');

/**
 * 댓글 작성 시 게시글 작성자에게 알림 생성 + 실시간 전송
 */
async function createCommentNotification({ boardAuthorId, senderId, senderName, boardId, boardTitle }) {
    if (boardAuthorId.toString() === senderId.toString()) return;

    const message = `"${boardTitle}" 게시글에 ${senderName}님이 댓글을 달았습니다.`;
    console.log(`[알림 생성] 수신자: ${boardAuthorId} / 발신자: ${senderId}`);

    const notification = await Notification.create({
        recipient: boardAuthorId,
        sender: senderId,
        board: boardId,
        message,
    });

    // Socket.io로 실시간 전송
    emitToUser(boardAuthorId.toString(), 'notification', {
        _id: notification._id,
        message,
        boardId,
        createdAt: notification.createdAt,
    });
}

/**
 * 특정 유저의 알림 목록 조회 (최근 20개)
 */
async function getNotifications(userId) {
    return Notification.find({ recipient: userId })
        .sort({ createdAt: -1 })
        .limit(20)
        .populate('sender', 'name')
        .populate('board', 'title');
}

/**
 * 읽지 않은 알림 수 조회
 */
async function getUnreadCount(userId) {
    return Notification.countDocuments({ recipient: userId, isRead: false });
}

/**
 * 알림 전체 읽음 처리
 */
async function markAllAsRead(userId) {
    await Notification.updateMany(
        { recipient: userId, isRead: false },
        { isRead: true }
    );
}

/**
 * 알림 단건 읽음 처리
 */
async function markAsRead(notificationId, userId) {
    await Notification.findOneAndUpdate(
        { _id: notificationId, recipient: userId },
        { isRead: true }
    );
}

module.exports = {
    createCommentNotification,
    getNotifications,
    getUnreadCount,
    markAllAsRead,
    markAsRead,
};
